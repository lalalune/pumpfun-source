const stock=(mint,fields={})=>({mint,symbol:'Stock '+mint,ticker:mint.toUpperCase(),...fields});
function setup(stocks=[]){
 const sentinel=Symbol.for('react.memo_cache_sentinel'),caches=new Map(),state=[],effects=[],calls=[];
 let stateCursor=0,effectCursor=0,debounced='',prefix='https://local.invalid/ipfs/',nsfw=false,query,supported={},raw=[];
 let result={isError:false,isFetching:false,data:undefined,error:undefined,refetch(){calls.push(['refetch']);return 'ignored';}};
 let translation=(key)=>key;const props={stocks,isLoading:false,onSelect(value){calls.push(['stock',value]);},onSelectPumpCoin(value){calls.push(['pump',value]);},onClose(...args){calls.push(['close',...args]);}};
 const jsx=(type,props,key)=>({type,props,key});
 const api=createCreateAssetPicker({React:{useState(initial){const i=stateCursor++;if(!(i in state))state[i]=initial;return[state[i],value=>{state[i]=value;}];},useEffect(effect,deps){const i=effectCursor++;if(!effects[i]||deps.some((v,j)=>v!==effects[i].deps[j]))effects[i]={effect,deps,pending:true};}},
  compiler(size){if(size===54||size===34){if(!caches.has(size))caches.set(size,Array(size).fill(sentinel));return caches.get(size);}return Array(size).fill(sentinel);},
  jsxRuntime:{jsx,jsxs:jsx,Fragment:'fragment'},useTranslation:()=>({t:translation}),useDebouncedValue(value,delay){calls.push(['debounce',value,delay]);return[debounced,{}];},
  useIpfsPrefix:()=>({ipfsPrefix:prefix}),useIncludeNsfw:()=>nsfw,useSupportedMints:()=>supported,useQuery(options){query=options;return result;},
  getContractClient(){calls.push(['client-factory']);return async(route,options)=>{calls.push(['request',route,options]);return raw;};},readCoins:readSearchCoins,normalizeCoins:normalizeSearchCoins,
  reportError(details){calls.push(['error',details]);},formatUsd(value){calls.push(['format',value]);return`USD:${value}`;},classes:(...values)=>values.join(' '),
  CategorySelector:'categories-port',StockArt:'stock-art-port',CoinAvatar:'avatar-port',CoinName:'coin-name-port',LoadError:'error-port',Skeleton:'skeleton-port',CloseIcon:'close-icon-port',SearchIcon:'search-icon-port'});
 function materialize(n){if(!n||typeof n!=='object')return n;if(Array.isArray(n))return n.map(materialize);if(typeof n.type==='function')return materialize(n.type(n.props));return {...n,props:{...n.props,children:materialize(n.props?.children)}};}
 function view(){stateCursor=0;effectCursor=0;const tree=materialize(api.Picker(props));for(const e of effects)if(e.pending){e.pending=false;e.effect();}return tree;}
 return{view,props,calls,state,query:()=>query,result(next){result={...result,...next};},settle(value){debounced=value;},prefix(value){prefix=value;},nsfw(value){nsfw=value;},supported(value){supported=value;},raw(value){raw=value;},language(value){translation=key=>value+key;}};
}
function nodes(n){if(Array.isArray(n))return n.flatMap(nodes);if(!n||typeof n!=='object')return[];return[n,...nodes(n.props?.children)];}
const all=(h)=>nodes(h.view());const input=(h)=>all(h).find(n=>n.type==='input');const category=(h)=>all(h).find(n=>n.type==='categories-port');
const rows=(h)=>all(h).filter(n=>n.type==='li'&&n.props?.children?.type==='button');
const text=(n)=>Array.isArray(n)?n.map(text).join(' '):n&&typeof n==='object'?text(n.props?.children):String(n??'');
const type=(h,value)=>input(h).props.onChange({target:{value}});
const selectCategory=(h,value)=>category(h).props.onValueChange(value);
