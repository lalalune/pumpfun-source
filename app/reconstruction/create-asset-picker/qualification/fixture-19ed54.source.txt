function useDebouncedValue(value,delay){
 const [current,set]=useState(value);
 useEffect(()=>{const started=performance.now();const id=setTimeout(()=>{timers.delete(id);calls.push(['debounce-settled',value,delay,performance.now()-started]);set(value);},delay);timers.add(id);return()=>{clearTimeout(id);timers.delete(id);};},[value,delay]);
 return[current,undefined];
}

}
